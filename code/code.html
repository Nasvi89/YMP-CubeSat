#include <Wire.h>
#include <Adafruit_BMP280.h>

Adafruit_BMP280 bmp;

void setup() {
  Serial.begin(9600);
  Serial.println("Инициализация систем CubeSat...");
  Wire.begin();
  if (!bmp.begin(0x76)) {
    Serial.println("Ошибка: BMP280 не найден!");
    while (1);
  }
  Serial.println("Системы готовы.");
}

void loop() {
  float t = bmp.readTemperature();
  float p = bmp.readPressure() / 100.0F;  // Па -> гПа
  Serial.print("Температура: "); Serial.print(t); Serial.println(" *C");
  Serial.print("Давление: ");    Serial.print(p); Serial.println(" hPa");
  Serial.println("-------------------------");
  delay(2000);
}

struct Telemetry {
  unsigned long time;   // мс с запуска
  float temperature;
  float pressure;
  bool sensorOk;
};

Telemetry readTelemetry() {
  Telemetry t;
  t.time = millis();
  t.temperature = bmp.readTemperature();
  t.pressure = bmp.readPressure() / 100.0F;
  t.sensorOk = !isnan(t.temperature) && !isnan(t.pressure);
  return t;
}

void sendTelemetry(const Telemetry& t) {
  Serial.print("TIME=");   Serial.print(t.time);
  Serial.print(";TEMP=");  Serial.print(t.temperature, 2);
  Serial.print(";PRESS="); Serial.print(t.pressure, 2);
  Serial.print(";SENSOR="); Serial.println(t.sensorOk ? "OK" : "ERROR");
}
